        ``` cpp
            int exgcd(int a, int b, int &x, int &y) 
            // 扩展欧几里得算法, 求x, y，使得ax + by = gcd(a, b)
            {
                if (!b)
                {
                    x = 1; y = 0;
                    return a;
                }
                //d始终都是a和b的最大公约数, 倒着传参是为了简化计算
                int d = exgcd(b, a % b, y, x);
                y -= (a / b) * x;
                return d;
                //返回d，使得能够在求最大公约数的同时完成x，y的凑整
                //若gcd(a, b)=1，x为a在模b意义下的逆元
            }

        ## 中国剩余定理

        中国剩余定理（CRT）用于求解同余方程组。

        模数互质（标准CRT）

        设模数 $m_1, m_2, ..., m_k$ 两两互质。

        - 令 $M = \prod m_i$
        - $M_i = M/m_i$

        由于 $M_i$ 与 $m_i$ 互质，存在逆元 $t_i$ 使 $M_i t_i \equiv 1 \pmod{m_i}$。

        **解为：**
        $$x \equiv \sum_{i=1}^{k} a_i M_i t_i \pmod{M}$$

        **验证：**
        对模 $m_j$，除第 $j$ 项为 $a_j$ 外，其余项均含 $m_j$ 因子而消去。

        ``` cpp
        typedef long long ll;
        // 扩展欧几里得算法
        ll exgcd(ll a, ll b, ll &x, ll &y);

        // 求逆元
        ll inv(ll a, ll mod) {
            ll x, y;
            exgcd(a, mod, x, y);
            return (x % mod + mod) % mod;
        }

        // 中国剩余定理（模数互质）
        // 输入：remainders余数数组，mods模数数组
        // 输出：最小非负解x，满足 x ≡ r[i] (mod m[i])
        ll CRT(vector<ll>& remainders, vector<ll>& mods) {
            int n = remainders.size();
            // 计算所有模数的乘积 M
            ll M = 1;
            for (int i = 0; i < n; i++) {
                M *= mods[i];
            }
            ll ans = 0;
            for (int i = 0; i < n; i++) {
                ll Mi = M / mods[i];       // M/mi
                ll ti = inv(Mi, mods[i]);  // Mi在模mi下的逆元
                ans = (ans + remainders[i] * Mi % M * ti % M) % M;
            }
            return ans;
        }
        ```

        ## 一阶线性同余方程
        ### 标准形式

        $$ax \equiv b \pmod{m}$$

        求解 $x$，或判断无解。

        ---

        ### 求解步骤
        #### 1. 化为不定方程

        $$ax \equiv b \pmod{m} \iff ax + my = b$$

        #### 2. 有解条件

        设 $d = \gcd(a, m)$，方程有解当且仅当：

        $$d \mid b$$

        #### 3. 求特解

        用扩展欧几里得解 $ax_0 + my_0 = d$，两边乘 $\frac{b}{d}$：

        $$x_1 = x_0 \cdot \frac{b}{d}$$

        #### 4. 求通解

        模 $m$ 意义下有 $d$ 个解：

        $$x \equiv x_1 + k \cdot \frac{m}{d} \pmod{m}, \quad k = 0, 1, ..., d-1$$

        **最小非负解：** $x_1 \bmod \frac{m}{d}$

        ---

        ## 模板代码

        ```cpp
        // 返回 (有解?, 最小非负解)
        pair<bool, ll> solve(ll a, ll b, ll m) {
            ll x, y;
            ll d = exgcd(a, m, x, y);
            if (b % d != 0) return {false, 0};
            x = x * (b / d) % m;
            ll mod = m / d;
            x = (x % mod + mod) % mod;
            return {true, x};
        }

        // 若需要全部 d 个解
        vector<ll> solve_all(ll a, ll b, ll m) {
            auto [ok, x0] = solve(a, b, m);
            if (!ok) return {};
            ll d = __gcd(a, m);
            vector<ll> res;
            ll mod = m / d;
            for (int k = 0; k < d; k++) {
                res.push_back((x0 + k * mod) % m);
            }
            return res;
        }

        ## 二次剩余
        ### 定义

        对于奇素数 $p$ 和整数 $a$（$p \nmid a$），若存在 $x$ 使得：

        $$x^2 \equiv a \pmod{p}$$

        则称 $a$ 是模 $p$ 的**二次剩余**，否则为**非二次剩余**。

        ---

        ### 勒让德符号

        $$\left(\frac{a}{p}\right) = 
        \begin{cases}
        1  & a \text{ 是二次剩余} \\
        -1 & a \text{ 是非二次剩余} \\
        0  & p \mid a
        \end{cases}$$

        ### 欧拉判别法

        $$\left(\frac{a}{p}\right) \equiv a^{\frac{p-1}{2}} \pmod{p}$$

        ---

        ### Cipolla 算法（求解二次剩余）

        **复杂度**：$O(\log p)$

        **步骤**：
        1. 若 $p=2$，$x \equiv a \pmod{2}$
        2. 随机找 $r$ 使 $r^2 - a$ 是非二次剩余
        3. 在扩域 $\mathbb{F}_p[\omega] = \{u + v\omega \mid \omega^2 = r^2 - a\}$ 中计算：
        $$x \equiv (r + \omega)^{\frac{p+1}{2}} \pmod{p}$$

        ### 模板代码

        ```cpp
        struct Complex {  // 扩域元素
            ll real, imag;  // real + imag * omega
            Complex(ll r = 0, ll i = 0) : real(r), imag(i) {}
        };

        ll w;  // omega^2 = w (即 r^2 - a)
        ll p;  // 模数

        Complex mul(const Complex& a, const Complex& b) {
            Complex res;
            res.real = (a.real * b.real + a.imag * b.imag % p * w) % p;
            res.imag = (a.real * b.imag + a.imag * b.real) % p;
            return res;
        }

        Complex qpow(Complex a, ll b) {
            Complex res(1, 0);
            while (b) {
                if (b & 1) res = mul(res, a);
                a = mul(a, a);
                b >>= 1;
            }
            return res;
        }

        ll qpow(ll a, ll b) {
            ll res = 1;
            while (b) {
                if (b & 1) res = res * a % p;
                a = a * a % p;
                b >>= 1;
            }
            return res;
        }

        // Cipolla 求 x^2 ≡ n (mod p)
        // 返回 (有解?, 解x)，另一个解为 p-x
        pair<bool, ll> cipolla(ll n, ll prime) {
            p = prime;
            if (n == 0) return {true, 0};
            if (qpow(n, (p - 1) / 2) != 1) return {false, 0}; 
            // 非二次剩余

            // 随机找 r 使 r^2 - n 是非二次剩余
            mt19937_64 rng(chrono::steady_clock::now().time_since_epoch().count());
            uniform_int_distribution<ll> dist(0, p - 1);
            ll r;
            do {
                r = dist(rng);
                w = (r * r % p - n + p) % p;
            } while (qpow(w, (p - 1) / 2) != p - 1);

            Complex base(r, 1);
            Complex ans = qpow(base, (p + 1) / 2);
            return {true, ans.real};
        }

        // 使用示例
        // auto [ok, x] = cipolla(a, p);
        // 另一解: p - x

        // 异或线性基（求最大值/最小值/k小值/排名）
        struct LinearBasis {
            static const int MAXL = 63;  // 处理64位整数
            long long basis[MAXL + 5];   // basis[i] 表示最高位为i的基
            int dim;                     // 线性基的维数（秩）
            
            LinearBasis() {
                memset(basis, 0, sizeof(basis));
                dim = 0;
            }
            
            // 插入一个数（标准高斯消元形式）
            bool insert(long long x) {
                for (int i = MAXL; i >= 0; i--) {
                    if (!(x >> i & 1)) continue;  // x的第i位为0就跳过
                    if (!basis[i]) {
                        basis[i] = x;
                        dim++;
                        return true;  // 插入成功
                    }
                    x ^= basis[i];
                }
                return false;  // 插入失败，x能被当前基表示
            }
            
            // 查询异或最大值（对0异或）
            long long queryMax() {
                long long res = 0;
                for (int i = MAXL; i >= 0; i--) {
                    if ((res ^ basis[i]) > res) {
                        res ^= basis[i];
                    }
                }
                return res;
            }
            
            // 查询异或最小值（非0的最小值）
            long long queryMin() {
                for (int i = 0; i <= MAXL; i++) {
                    if (basis[i]) return basis[i];
                }
                return 0;
            }
            
            // 判断一个数能否被线性基表示
            bool contains(long long x) {
                for (int i = MAXL; i >= 0; i--) {
                    if (x >> i & 1) {
                        if (!basis[i]) return false;
                        x ^= basis[i];
                    }
                }
                return true;
            }

- # 图论：
    1. ## Tarjan

        ### 算法思想

        对有向图进行强连通分量（SCC）缩点，缩点后得到 DAG。

        **核心数组：**
        - `dfn[u]`：DFS 序
        - `low[u]`：能回溯到的最早 dfn
        - `in_stk[u]`：是否在栈中
        - `scc[u]`：所属 SCC 编号

        **判定 SCC：** `dfn[u] == low[u]` 时，栈顶到 `u` 的所有点构成一个 SCC。

        ---

        ## 模板代码

        ```cpp
        const int N = 1e5 + 10, M = 2e5 + 10;

        int n, m;
        vector<int> g[N];          // 原图

        int dfn[N], low[N], timer;
        int stk[N], top;
        bool in_stk[N];
        int scc[N], scc_cnt;       // scc[i] = i 所属 SCC 编号
        int sz[N];                 // sz[i] = SCC i 的大小

        void tarjan(int u) {
            dfn[u] = low[u] = ++timer;
            stk[++top] = u;
            in_stk[u] = true;

            for (int v : g[u]) {
                if (!dfn[v]) {
                    tarjan(v);
                    low[u] = min(low[u], low[v]);
                } else if (in_stk[v]) {
                    low[u] = min(low[u], dfn[v]);
                }
            }

            // 找到一个 SCC
            if (dfn[u] == low[u]) {
                scc_cnt++;
                int v;
                do {
                    v = stk[top--];
                    in_stk[v] = false;
                    scc[v] = scc_cnt;
                    sz[scc_cnt]++;
                } while (v != u);
            }
        }

        // 缩点建新图
        vector<int> ng[N];         // DAG
        int in[N];                 // 入度（拓扑排序用）

        void build_dag() {
            for (int i = 1; i <= n; i++) {
                if (!dfn[i]) tarjan(i);  // 不连通时
            }
            for (int u = 1; u <= n; u++) {
                for (int v : g[u]) {
                    if (scc[u] != scc[v]) {
                        ng[scc[u]].push_back(scc[v]);
                        in[scc[v]]++;
                    }
                }
            }
            // 可选：DAG 去重边
            for (int i = 1; i <= scc_cnt; i++) {
                sort(ng[i].begin(), ng[i].end());
                ng[i].erase(unique(ng[i].begin(), ng[i].end()), ng[i].end());
            }
        }
        ```

        constexpr int maxn = 205;
        constexpr int INF = 1e18;
        struct e {
            int to,rev;
            int c;
        };
        vector<e> g[maxn];
        int lv[maxn],cur[maxn];
        int n,m,s,t;
        void adde(int u,int v, int c) {
            g[u].push_back({v, (int)g[v].size(), c});
            g[v].push_back({u, (int)g[u].size()-1, 0});
        }
        bool bfs() {
            memset(lv,-1,sizeof(lv));
            queue<int> q;
            q.push(s);
            lv[s]=0;
            while(!q.empty()) {
                int u=q.front();
                q.pop();
                for (e& ei:g[u]) {
                    if (ei.c>0 && lv[ei.to]==-1) {
                        lv[ei.to]=lv[u]+1;
                        q.push(ei.to);
                    }
                }
            }
            return lv[t] != -1;
        }
        int dfs(int u, int sum) {
            if (u==t) return sum;
            int k,res=0;
            for (int i=cur[u];i<g[u].size() && sum;i++) {
                cur[u] = i;
                e& ei = g[u][i];
                if (ei.c>0 && lv[ei.to]==lv[u]+1) {
                    k = dfs(ei.to,min(sum, ei.c));
                    if (k==0) {
                        lv[ei.to] = -1;
                        continue;
                    }
                    ei.c-=k;
                    g[ei.to][ei.rev].c+=k;
                    res+=k;
                    sum-=k;
                }
            }
            return res;
        }
        int Dinic() {
            int res = 0;
            while (bfs()) {
                memset(cur, 0, sizeof(cur));
                res += dfs(s,INF);
            }
            return res;
        }

        void dfs(int u, int fa) {
            dp1[u] = dp2[u] = 0;
            for (auto [v, w] : g[u]) {
                if (v == fa) continue;
                dfs(v, u);
                ll d = dp1[v] + w;
                if (d > dp1[u]) {
                    dp2[u] = dp1[u];
                    dp1[u] = d;
                } 
                else if (d > dp2[u]) {
                    dp2[u] = d;
                }
            }
            ans = max(ans, dp1[u] + dp2[u]);
        }

            ans = 0;
            dfs(1, 0);

        ``` cpp
        int n,m,s,t;
        struct e {
            int t,w,c;
            int rev;
        };
        vector<e> g[maxn];
        array<int,maxn> dist,cur;
        array<bool, maxn> vis;
        void addEdge(int u,int v, int w, int c) {
            g[u].push_back({v,w,c, (int)g[v].size()});
            g[v].push_back({u,0,-c, (int)g[u].size()-1});
        }
        bool spfa() {
            //spfa寻找最短路并建立分层图
            dist.fill(1e9);
            array<bool,maxn> inq{};
            queue<int> q;
            dist[s] = 0;
            inq[s] = true;
            q.push(s);
            while (!q.empty()) {
                int u = q.front();
                q.pop();
                inq[u] = false;
                for (e& ei:g[u]) {
                    if (ei.w && dist[u]+ei.c < dist[ei.t]) {
                        dist[ei.t] = dist[u]+ei.c;
                        if (!inq[ei.t]) {
                            inq[ei.t] = true;
                            q.push(ei.t);
                        }
                    }
                }
            }
            return dist[t] < 1e9;
        }
        int dfs(int u, int lim) {
            if (u==t) return lim;
            vis[u] = true;
            int res = 0;
            int k;
            for (int i=cur[u];i<g[u].size() && lim;++i) {
                cur[u] = i;
                e& ei = g[u][i];
                if (ei.w && dist[u]+ei.c==dist[ei.t] && !vis[ei.t]) {
                    k = dfs(ei.t, min(lim, ei.w));
                    if (!k) {
                        dist[ei.t] = -1;
                        continue;
                    }
                    ei.w-=k;
                    g[ei.t][ei.rev].w+=k;
                    res += k;
                    lim -= k;
                }
            }
            vis[u] = false;
            return res;
        }

                addEdge(u,v,w,c);

            if(l==r){
                tr[u].sum = a[l];
                return;
            }
            int mid = (l+r)/2;
            build(u*2, l, mid);
            build(u*2+1, mid+1, r);
            pushup(u);
        }
        void pushdown(int u){
            if(tr[u].lz){
                int lazy = tr[u].lz;
                tr[u*2].sum += lazy * (tr[u*2].r-tr[u*2].l+1);
                tr[u*2].lz+=lazy;
                tr[u*2+1].sum += lazy * (tr[u*2+1].r-tr[u*2+1].l+1);
                tr[u*2+1].lz+=lazy;
            }
            tr[u].lz = 0;
        } 
        void modify(int u, int l, int r, int d){
            if(tr[u].l>=l && tr[u].r<=r){
                tr[u].sum += d * (tr[u].r-tr[u].l+1);
                tr[u].lz += d;
                return;
            }
            pushdown(u); 
            int mid = (tr[u].l+tr[u].r)/2;
            if(l<=mid) modify(u*2, l, r, d);
            if(r>mid) modify(u*2+1, l, r, d);
            pushup(u);
        }
        int query(int u, int l, int r){
            if(tr[u].l>=l && tr[u].r<=r){
                return tr[u].sum;
            }
            pushdown(u);
            int res = 0;
            int mid = (tr[u].l+tr[u].r)/2;
            if(l<=mid) res += query(u*2, l, r);
            if(r>mid) res += query(u*2+1, l, r);
            return res;
        }

        ``` cpp
        int n,m;
        array<int,maxn> lc,rc,dist,rt;
        //左子节点，右子节点，Npl值，根
        bool tf[maxn];//是否被删除
        struct node {
            int id,v;
            bool operator < (const node &p) const {
                return v==p.v ? id<p.id : v<p.v;
            }
        };
        node v[maxn];
        int find(int x) {
            if (x==rt[x]) return x;
            return rt[x] = find(rt[x]);//查找根节点
        }
        int merge(int x, int y) {
            //递归合并，将右子树与y堆合并
            if (!x || !y) return x+y;
            if (v[y]<v[x]) swap(x,y);
            rc[x] = merge(rc[x], y);
            if (dist[lc[x]]<dist[rc[x]]) swap(lc[x],rc[x]);
            //保证左偏性
            dist[x] = dist[rc[x]]+1;//更新Npl
            return x;
        }
        int main() {
            dist[0] = -1;
            cin >> n >> m;
            for (int i=1;i<=n;++i) {
                int vi;
                cin >> vi;
                rt[i]=i;
                v[i].id=i;
                v[i].v=vi;
            }
            while (m--) {
                int op;
                cin >> op;
                if (op==1) {
                    int x,y;
                    cin >> x >> y;
                    if (tf[x] || tf[y]) continue;
                    x = find(x);
                    y = find(y);
                    if (x!=y) rt[x] = rt[y] = merge(x,y);
                }
                else {
                    int x;
                    cin >> x;
                    if (tf[x]) {
                        cout << "-1\n";
                        continue;
                    }
                    x = find(x);
                    cout << v[x].v << "\n";
                    tf[x] = true;
                    rt[lc[x]] = rt[rc[x]] = rt[x]=merge(lc[x],rc[x]);
                    //通过合并左右子树来删除堆的最小值，注意要记录x的根保证后续正确查询
                    lc[x] = rc[x] = dist[x] = 0;
                }
            }
            return 0;
        }
